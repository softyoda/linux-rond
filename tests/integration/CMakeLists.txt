cmake_minimum_required(VERSION 3.22)
project(linux_rond_integration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBEI REQUIRED IMPORTED_TARGET libei-1.0)
add_executable(round_probe probe.cpp)
target_link_libraries(round_probe PRIVATE Qt6::Widgets)
add_executable(round_input_driver input-driver.cpp)
target_link_libraries(round_input_driver PRIVATE Qt6::DBus PkgConfig::LIBEI)
add_executable(round_map_point map-point.cpp ../../kwin/round-workspace/geometry.cpp)
target_include_directories(round_map_point PRIVATE ../../kwin/round-workspace)
target_link_libraries(round_map_point PRIVATE Qt6::Core)
add_executable(round_demo_control demo-control.cpp)
set_target_properties(round_demo_control PROPERTIES AUTOMOC ON)
target_link_libraries(round_demo_control PRIVATE Qt6::DBus)

cmake_minimum_required(VERSION 3.22)
project(linux_rond_workspace LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KWin 6.8 REQUIRED)
add_library(linux_rond_workspace MODULE main.cpp workspace.cpp geometry.cpp input.cpp overview.cpp resources.qrc)
set_target_properties(linux_rond_workspace PROPERTIES PREFIX "")
target_link_libraries(linux_rond_workspace PRIVATE KWin::kwin)
target_compile_options(linux_rond_workspace PRIVATE -Wall -Wextra)
add_library(linux_rond_workspace_loader MODULE loader.cpp)
set_target_properties(linux_rond_workspace_loader PROPERTIES PREFIX "")
target_link_libraries(linux_rond_workspace_loader PRIVATE KWin::kwin ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(linux_rond_workspace PRIVATE "LINKER:--no-undefined")
    target_link_options(linux_rond_workspace_loader PRIVATE "LINKER:--no-undefined")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(geometry_test geometry_test.cpp geometry.cpp resources.qrc)
    target_link_libraries(geometry_test PRIVATE Qt6::Core)
    set_target_properties(geometry_test PROPERTIES AUTOMOC OFF)
    add_test(NAME projection_geometry COMMAND geometry_test)
    add_executable(overview_test overview_test.cpp overview.cpp)
    target_link_libraries(overview_test PRIVATE Qt6::Core Qt6::Gui)
    set_target_properties(overview_test PROPERTIES AUTOMOC OFF)
    add_test(NAME overview_layout COMMAND overview_test)
    add_executable(inputcoordinates_test inputcoordinates_test.cpp)
    target_link_libraries(inputcoordinates_test PRIVATE Qt6::Core)
    set_target_properties(inputcoordinates_test PROPERTIES AUTOMOC OFF)
    add_test(NAME touch_coordinates COMMAND inputcoordinates_test)
endif()

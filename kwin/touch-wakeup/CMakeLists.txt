cmake_minimum_required(VERSION 3.22)
project(linux_rond_touch_wakeup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KWin 6.8 REQUIRED)
add_library(linux_rond_touch_wakeup MODULE touchwakeup.cpp)
set_target_properties(linux_rond_touch_wakeup PROPERTIES PREFIX "")
target_link_libraries(linux_rond_touch_wakeup PRIVATE KWin::kwin)
include(CTest)
if(BUILD_TESTING)
  add_executable(tapstate_test tapstate_test.cpp)
  set_target_properties(tapstate_test PROPERTIES AUTOMOC OFF)
  add_test(NAME tapstate COMMAND tapstate_test)
endif()
